-- Consultar todas las citas del paciente con id_paciente = 1
SELECT 
    id_cita,
    nombre_medico,
    especialidad,
    fecha,
    hora,
    motivo,
    estado,
    IF(permite_reprogramar_cancelar = 1, 'Sí', 'No (Menos de 24h)') AS puede_modificar
FROM vw_agenda_paciente
WHERE id_paciente = 1
ORDER BY fecha DESC, hora DESC;

----------------------------------------------------------------------------

-- Consultar las citas del día para el Dr. Torres (id_medico = 1)
SELECT 
    id_cita,
    hora,
    nombre_paciente,
    dni_paciente,
    edad_paciente,
    genero_paciente,
    tipo_sangre,
    motivo,
    estado
FROM vw_agenda_medico
WHERE id_medico = 1 
  AND fecha = CURDATE()
ORDER BY hora ASC;

----------------------------------------------------------------------------

-- Listar horarios de atención para Cardiología
SELECT 
    nombre_medico,
    especialidad,
    dia_semana_nombre,
    hora_inicio,
    hora_fin,
    tipo
FROM vw_medicos_horarios
WHERE especialidad = 'Cardiología' 
  AND horario_activo = 1
ORDER BY dia_semana ASC, hora_inicio ASC;

----------------------------------------------------------------------------

-- Obtener correos y teléfonos de pacientes que aceptan recordatorios de 24h
SELECT 
    id_cita,
    nombre_paciente,
    email_paciente,
    telefono_paciente,
    nombre_medico,
    especialidad,
    fecha,
    hora,
    notif_email,
    notif_whatsapp
FROM vw_recordatorios_pendientes
WHERE recordatorio_24h = 1;

----------------------------------------------------------------------------

-- A. Contar total de citas agrupadas por Estado
SELECT 
    estado, 
    COUNT(*) AS total
FROM vw_reporte_general_citas
GROUP BY estado;

-- B. Contar citas reservadas por Especialidad
SELECT 
    especialidad, 
    COUNT(*) AS total_reservas
FROM vw_reporte_general_citas
GROUP BY especialidad
ORDER BY total_reservas DESC;
