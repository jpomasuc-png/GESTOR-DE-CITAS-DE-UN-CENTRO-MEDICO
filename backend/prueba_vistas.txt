USE sistema_citas;

-- ====================================================================
-- PRUEBAS DE VISTAS
-- ====================================================================

-- ---------------------------------------------------------
-- 1. Vista: vw_agenda_paciente
-- ---------------------------------------------------------
-- Consultar todas las citas del paciente con id_paciente = 1
SELECT
    id_cita,
    nombre_medico,
    especialidad,
    fecha,
    hora,
    motivo,
    estado,
    IF(permite_reprogramar_cancelar = 1, 'Sí', 'No (Menos de 24h)') AS puede_modificar
FROM vw_agenda_paciente
WHERE id_paciente = 1
ORDER BY fecha DESC, hora DESC;

-- ---------------------------------------------------------
-- 2. Vista: vw_agenda_medico
-- ---------------------------------------------------------
-- Consultar las citas del día para el Dr. Torres
SELECT
    id_cita,
    hora,
    nombre_paciente,
    dni_paciente,
    edad_paciente,
    genero_paciente,
    tipo_sangre,
    motivo,
    estado
FROM vw_agenda_medico
WHERE id_medico = 1
  AND fecha = CURDATE()
ORDER BY hora ASC;

-- ---------------------------------------------------------
-- 3. Vista: vw_medicos_horarios
-- ---------------------------------------------------------
-- Mostrar horarios del médico de Cardiología
SELECT
    nombre_medico,
    especialidad,
    dia_semana_nombre,
    hora_inicio,
    hora_fin,
    tipo
FROM vw_medicos_horarios
WHERE especialidad = 'Cardiología'
  AND horario_activo = 1
ORDER BY dia_semana ASC, hora_inicio ASC;

-- ---------------------------------------------------------
-- 4. Vista: vw_recordatorios_pendientes
-- ---------------------------------------------------------
-- Obtener correos y teléfonos de pacientes con recordatorios activos
SELECT
    id_cita,
    nombre_paciente,
    email_paciente,
    telefono_paciente,
    nombre_medico,
    especialidad,
    fecha,
    hora,
    notif_email,
    notif_whatsapp
FROM vw_recordatorios_pendientes
WHERE recordatorio_24h = 1;

-- ---------------------------------------------------------
-- 5. Vista: vw_reporte_general_citas
-- ---------------------------------------------------------
-- A. Contar citas por estado
SELECT
    estado,
    COUNT(*) AS total
FROM vw_reporte_general_citas
GROUP BY estado;

-- B. Contar reservas por especialidad
SELECT
    especialidad,
    COUNT(*) AS total_reservas
FROM vw_reporte_general_citas
GROUP BY especialidad
ORDER BY total_reservas DESC;

-- ---------------------------------------------------------
-- 6. Vista: vw_vista_medico_pacientes
-- ---------------------------------------------------------
-- Ver pacientes asociados a un médico
SELECT *
FROM vw_vista_medico_pacientes
WHERE id_medico = 1
ORDER BY nombre_paciente;

-- ---------------------------------------------------------
-- 7. Vista: vw_vista_admin_medicosypacientes
-- ---------------------------------------------------------
-- Ver médicos y pacientes para administración
SELECT *
FROM vw_vista_admin_medicosypacientes
ORDER BY rol, nombre;
