-- FUNCIÓN: fn_calcular_edad
SELECT
    p.id_paciente,
    u.nombre,
    p.fecha_nacimiento,
    fn_calcular_edad(p.fecha_nacimiento) AS edad_actual
FROM pacientes p
INNER JOIN usuarios u ON p.id_usuario = u.id_usuario;

-- FUNCIÓN: fn_validar_anticipacion_24h
-- Debe devolver 1 si la cita tiene al menos 24 horas de anticipación
SELECT fn_validar_anticipacion_24h('2026-12-31', '10:00:00') AS es_valido_futuro;

-- Debe devolver 0 si falta menos de 24 horas
SELECT fn_validar_anticipacion_24h(CURDATE(), CURTIME()) AS es_valido_hoy;
